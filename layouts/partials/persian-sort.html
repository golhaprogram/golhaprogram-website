{{- /* Sort taxonomy terms in Persian alphabetical order (full name, not just the first letter). Returns a slice of dicts: term, pages, order. */ -}} {{- $terms := slice -}} {{- range $term, $pages := .Data.Terms -}} {{- $name := $term -}} {{- $terms = $terms | append (dict "term" $term "pages" $pages "order" (partial "fa-sort-key" $name)) -}} {{- end -}} {{- return (sort $terms "order") -}}
